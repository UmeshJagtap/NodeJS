Steps
1. Client sends request to Server
2. Server checks Redis for data
3a. If Cache Hit --> retyrn data to client
3b. If Cache Miss--> fetch data from DB
4. DB returns data to Server
5. Server stores data in Redis (cache)
6. Server sends response to Client

Redis Defination
Redis is an in-memory key-value database used for super fast data access.

+ index.js
import express from 'express';
import dotenv from 'dotenv';
import connectDb from './config/db.js';
import User from './model/user.model.js';
dotenv.config();

const port = process.env.PORT || 5000;

const app = express();

app.use(express.json());

app.get('/', (req, res) => {
  return res.status(200).json({ message: 'hello from redis-- 2026' });
});

app.post('/create', async (req, res) => {
  const { name, email, password } = req.body;
  console.log('req.body', req.body);
  const user = await User.create({
    name,
    email,
    password,
  });
  // name, email, password,
  // name: 'umesh',
  // email: 'qwerty' + Date.now() + '@gmail.com',
  // password: 'try123*()',
  return res.status(200).json(user, { message: 'user created successfully' });
});

app.get('/get', async (req, res) => {
  const users = await User.find({});
  return res.json(users, { message: 'users fetched successfully' });
});

app.get('/get-with-redis', async (req, res) => {
  const cached = redis.get('user:all');

  if (cached) {
    const users = JSON.parse(cached);
    return res.json(users, {
      message: 'users fetched successfully from cache',
    });
  }

  const users = await User.find({});
  await redis.set('user:all', JSON.stringify(users), 'EX', 10); // expire in 10 seconds

  return res.json(JSON.parse(users));
});

app.listen(port, () => {
  connectDb();
  console.log(`server started ${port}`);
});

Queues :
Queues ek waiting line hoti hai jisme backend tasks store hote hain
taaki unhe background me process kiya ja sake 
bina user ko wait karvaye 

Suppose :
signup API hai

SignUp ke baad:
> welcome email
> notification
> analytics
karna hai

Normal Flow : ...aka aam zindagi
signup API :- name, email, password --> create user in db
  --> send signup email --> signup successfully

Queues Flow : ...aka mentos zindagi
signup API :- name, email, password --> create user in db
  --> { send signup email (queue) } --> signup successfully

[+] worker.js
import { Queue, Worker } from 'bullmq';
import Redis from 'ioredis';
import sendEmail from './lib/sendEmail.js';

const connection = new Redis(
  process.env.REDIS_URL || 'redis://localhost:6379',
  {
    maxRetriesPerRequest: null,
  }
);

const worker = new Worker(
  'emailQueue',
  async (job) => {
    console.log('Job started for email:', email);
    const email = job.data.email;
    await sendEmail(email);
    console.log('Job completed for email:', email);
  },
  { connection }
);

export default worker;
